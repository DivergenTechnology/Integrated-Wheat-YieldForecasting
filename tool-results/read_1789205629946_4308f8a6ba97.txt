   374→interface IndexFieldProps {
   375→  label: string
   376→  hint: string
   377→  min: number
   378→  max: number
   379→  step: number
   380→  value: number
   381→  decimals: number
   382→  onChange: (v: number) => void
   383→  unit?: string
   384→}
   385→
   386→function IndexField({ label, hint, min, max, step, value, decimals, onChange, unit }: IndexFieldProps) {
   387→  return (
   388→    <div className="space-y-1.5">
   389→      <div className="flex items-center justify-between">
   390→        <Label className="text-sm font-medium">{label}</Label>
   391→        <div className="flex items-center gap-1.5">
   392→          <Input
   393→            type="number"
   394→            min={min}
   395→            max={max}
   396→            step={step}
   397→            value={value}
   398→            onChange={(e) => {
   399→              const v = Number(e.target.value)
   400→              if (Number.isFinite(v)) onChange(v)
   401→            }}
   402→            className="h-8 w-24 text-right tabular-nums"
   403→            aria-label={`${label} value`}
   404→          />
   405→          {unit && <span className="text-xs text-stone-500">{unit}</span>}
   406→        </div>
   407→      </div>
   408→      <Slider
   409→        min={min}
   410→        max={max}
   411→        step={step}
   412→        value={[value]}
   413→        onValueChange={(vals) => onChange(vals[0])}
   414→        aria-label={`${label} slider`}
   415→      />
   416→      <p className="text-xs text-stone-500">{hint}</p>
   417→    </div>
   418→  )
   419→}
   420→
   421→export function AddReadingDialog({
   422→  zones,
   423→  defaultZoneId,
   424→  open,
   425→  onOpenChange,
   426→  onAdded,
   427→}: {
   428→  zones: { id: string; name: string }[]
   429→  defaultZoneId?: string
   430→  open: boolean
   431→  onOpenChange: (v: boolean) => void
   432→  onAdded: () => void
   433→}) {
   434→  const { toast } = useToast()
   435→  const [zoneId, setZoneId] = useState(defaultZoneId ?? zones[0]?.id ?? '')
   436→  const [surveyDate, setSurveyDate] = useState(new Date().toISOString().slice(0, 10))
   437→  const [growthStage, setGrowthStage] = useState<GrowthStage>('mid')
   438→  const [ndvi, setNdvi] = useState(0.75)
   439→  const [gndvi, setGndvi] = useState(0.68)
   440→  const [ndre, setNdre] = useState(0.32)
   441→  const [lai, setLai] = useState(3.5)
   442→  const [canopyCover, setCanopyCover] = useState(75)
   443→  const [droneModel, setDroneModel] = useState('DJI Mavic 3M')
   444→  const [notes, setNotes] = useState('')
   445→  const [saving, setSaving] = useState(false)
   446→
   447→  function resetDefaults() {
   448→    setSurveyDate(new Date().toISOString().slice(0, 10))
   449→    setGrowthStage('mid')
   450→    setNdvi(0.75)
   451→    setGndvi(0.68)
   452→    setNdre(0.32)
   453→    setLai(3.5)
   454→    setCanopyCover(75)
   455→    setNotes('')
   456→  }
   457→
   458→  async function submit() {
   459→    if (!zoneId) {
   460→      toast({ title: 'Select a zone', description: 'Choose which farming zone this survey belongs to.', variant: 'destructive' })
   461→      return
   462→    }
   463→    setSaving(true)
   464→    try {
   465→      const res = await fetch('/api/readings', {
   466→        method: 'POST',
   467→        headers: { 'Content-Type': 'application/json' },
   468→        body: JSON.stringify({ zoneId, surveyDate, growthStage, ndvi, gndvi, ndre, lai, canopyCover, droneModel, notes }),
   469→      })
   470→      const json = await res.json()
   471→      if (!res.ok) throw new Error(json.error || 'Failed to save survey')
   472→      toast({ title: 'UAV survey recorded', description: 'Yield forecasts have been recomputed for the zone.' })
   473→      resetDefaults()
   474→      onOpenChange(false)
   475→      onAdded()
   476→    } catch (e) {
   477→      toast({
   478→        title: 'Error',
   479→        description: e instanceof Error ? e.message : 'Failed to save survey',
   480→        variant: 'destructive',
   481→      })
   482→    } finally {
   483→      setSaving(false)
   484→    }
   485→  }
   486→
   487→  return (
   488→    <Dialog open={open} onOpenChange={onOpenChange}>
   489→      <DialogContent className="max-w-xl max-h-[90vh] overflow-y-auto">
   490→        <DialogHeader>
   491→          <DialogTitle className="flex items-center gap-2">
   492→            <Plane className="h-5 w-5 text-emerald-700" aria-hidden /> Record UAV survey
   493→          </DialogTitle>
   494→          <DialogDescription>
   495→            Enter vegetation index values extracted from the drone multispectral flight.
   496→          </DialogDescription>
   497→        </DialogHeader>
   498→
   499→        <div className="grid gap-4 py-1">
   500→          <div className="grid gap-4 sm:grid-cols-3">
   501→            <div className="space-y-1.5 sm:col-span-2">
   502→              <Label>Farming zone</Label>
   503→              <Select value={zoneId} onValueChange={setZoneId}>
   504→                <SelectTrigger aria-label="Farming zone">
   505→                  <SelectValue placeholder="Select zone" />
   506→                </SelectTrigger>
   507→                <SelectContent>
   508→                  {zones.map((z) => (
   509→                    <SelectItem key={z.id} value={z.id}>{z.name}</SelectItem>
   510→                  ))}
   511→                </SelectContent>
   512→              </Select>
   513→            </div>
   514→            <div className="space-y-1.5">
   515→              <Label>Survey date</Label>
   516→              <Input type="date" value={surveyDate} onChange={(e) => setSurveyDate(e.target.value)} aria-label="Survey date" />
   517→            </div>
   518→          </div>
   519→
   520→          <div className="space-y-1.5">
   521→            <Label>Growth stage</Label>
   522→            <Select value={growthStage} onValueChange={(v) => setGrowthStage(v as GrowthStage)}>

   563→/* ------------------------------------------------------------------ */
   564→
   565→export function AddZoneDialog({
   566→  open,
   567→  onOpenChange,
   568→  onAdded,
   569→}: {
   570→  open: boolean
   571→  onOpenChange: (v: boolean) => void
   572→  onAdded: () => void
   573→}) {
   574→  const { toast } = useToast()
   575→  const [name, setName] = useState('')
   576→  const [cropType, setCropType] = useState('maize')
   577→  const [area, setArea] = useState('10')
   578→  const [soilType, setSoilType] = useState('')
   579→  const [description, setDescription] = useState('')
   580→  const [saving, setSaving] = useState(false)
   581→
   582→  async function submit() {
   583→    if (!name.trim()) {
   584→      toast({ title: 'Name required', description: 'Give the farming zone a name.', variant: 'destructive' })
   585→      return
   586→    }
   587→    const areaNum = Number(area)
   588→    if (!Number.isFinite(areaNum) || areaNum <= 0) {
   589→      toast({ title: 'Invalid area', description: 'Area must be a positive number of hectares.', variant: 'destructive' })
   590→      return
   591→    }
   592→    setSaving(true)
   593→    try {
   594→      const res = await fetch('/api/zones', {
   595→        method: 'POST',
   596→        headers: { 'Content-Type': 'application/json' },
   597→        body: JSON.stringify({ name, cropType, areaHectares: areaNum, soilType, description }),
   598→      })
   599→      const json = await res.json()
   600→      if (!res.ok) throw new Error(json.error || 'Failed to create zone')
   601→      toast({ title: 'Zone created', description: `${name} is ready — record a UAV survey to forecast yield.` })
   602→      setName('')
   603→      setSoilType('')
   604→      setDescription('')
   605→      onOpenChange(false)
   606→      onAdded()
   607→    } catch (e) {
   608→      toast({
   609→        title: 'Error',
   610→        description: e instanceof Error ? e.message : 'Failed to create zone',
   611→        variant: 'destructive',
   612→      })
   613→    } finally {
   614→      setSaving(false)
   615→    }
   616→  }
   617→
   618→  return (
   619→    <Dialog open={open} onOpenChange={onOpenChange}>
   620→      <DialogContent className="max-w-md">
   621→        <DialogHeader>
   622→          <DialogTitle>Add farming zone</DialogTitle>
   623→          <DialogDescription>Register a new zone, then record UAV surveys to forecast its yield.</DialogDescription>
   624→        </DialogHeader>
   625→        <div className="grid gap-4 py-1">
   626→          <div className="space-y-1.5">
   627→            <Label htmlFor="zone-name">Zone name</Label>
   628→            <Input id="zone-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. North Block C" />
   629→          </div>
   630→          <div className="grid grid-cols-2 gap-4">
   631→            <div className="space-y-1.5">
   632→              <Label>Crop</Label>
   633→              <Select value={cropType} onValueChange={setCropType}>
   634→                <SelectTrigger aria-label="Crop type">
   635→                  <SelectValue />
   636→                </SelectTrigger>
   637→                <SelectContent>
   638→                  {Object.entries(CROP_PROFILES).map(([key, p]) => (
   639→                    <SelectItem key={key} value={key}>{p.emoji} {p.label}</SelectItem>
   640→                  ))}
   641→                </SelectContent>
   642→              </Select>
   643→            </div>
   644→            <div className="space-y-1.5">
   645→              <Label htmlFor="zone-area">Area (ha)</Label>
   646→              <Input id="zone-area" type="number" min="0.1" step="0.1" value={area} onChange={(e) => setArea(e.target.value)} />
   647→            </div>
   648→          </div>
   649→          <div className="space-y-1.5">
   650→            <Label htmlFor="zone-soil">Soil type</Label>
   651→            <Input id="zone-soil" value={soilType} onChange={(e) => setSoilType(e.target.value)} placeholder="e.g. Sandy loam" />
   652→          </div>
   653→          <div className="space-y-1.5">
   654→            <Label htmlFor="zone-desc">Description</Label>
   655→            <Textarea id="zone-desc" value={description} onChange={(e) => setDescription(e.target.value)} rows={2} placeholder="Irrigation, fertility history…" />
   656→          </div>
   657→        </div>
   658→        <DialogFooter>
   659→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   660→          <Button onClick={submit} disabled={saving} className="bg-emerald-700 hover:bg-emerald-800">
   661→            {saving ? 'Creating…' : 'Create zone'}
   662→          </Button>
   663→        </DialogFooter>
   664→      </DialogContent>
   665→    </Dialog>
   666→  )
   667→}
